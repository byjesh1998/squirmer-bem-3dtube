cmake_minimum_required(VERSION 3.16)
project(squirmer_bem CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O3 -march=native")

# BLAS + LAPACK (OpenBLAS recommended: it provides both, and is fast)
set(BLA_VENDOR OpenBLAS CACHE STRING "BLAS vendor")
find_package(BLAS)
find_package(LAPACK)
if(NOT BLAS_FOUND OR NOT LAPACK_FOUND)
  unset(BLA_VENDOR CACHE)
  set(BLA_VENDOR "")
  find_package(BLAS REQUIRED)
  find_package(LAPACK REQUIRED)
endif()
set(LINALG_LIBS ${LAPACK_LIBRARIES} ${BLAS_LIBRARIES})

include_directories(${CMAKE_SOURCE_DIR}/src)

add_executable(squirmer_bem src/main.cpp)
target_link_libraries(squirmer_bem ${LINALG_LIBS})

# ---- tests (run from the repository root so outputs go to outputs/tests)
enable_testing()
foreach(t test_duct_flow test_squirmer test_trajectory)
  add_executable(${t} tests/${t}.cpp)
  target_link_libraries(${t} ${LINALG_LIBS})
endforeach()
add_test(NAME duct_flow  COMMAND test_duct_flow  WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_test(NAME squirmer   COMMAND test_squirmer   WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
add_test(NAME trajectory COMMAND test_trajectory WORKING_DIRECTORY ${CMAKE_SOURCE_DIR})
